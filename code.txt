import WrapperApp from './nodewrapper.js';
import { createServer } from 'node:http';

const app = new WrapperApp();

app.get('/posts', (req, res) => {
  res.writeHead(200, { 'Content-Type': 'text/plain' });
  res.end('All your posts!\n');
});
app.get('/posts/first', (req, res) => {
  res.writeHead(200, { 'Content-Type': 'text/plain' });
  res.end('THis is the first post!\n');
});

// Then handle requests
const server = createServer((req, res) => {
  app.handleRequest(req, res); // Actually USE your router

  // Optional: send 404 if no route matched
  if (!res.headersSent) {
    res.end('404 page');
  }
});

// starts a simple http server locally on port 3000
server.listen(3000, '127.0.0.1', () => {
  console.log('Listening on 127.0.0.1:3000');
});
